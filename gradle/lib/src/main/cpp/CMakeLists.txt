cmake_minimum_required(VERSION 3.22.1)
project("ai-chat" LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LLAMA_SRC ${CMAKE_CURRENT_LIST_DIR}/../../../../../third_party/llama.cpp)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
add_subdirectory(${LLAMA_SRC} build-llama)
add_library(${CMAKE_PROJECT_NAME} SHARED ai_chat.cpp)
target_compile_definitions(${CMAKE_PROJECT_NAME} PRIVATE GGML_SYSTEM_ARCH=ARM GGML_CPU_KLEIDIAI=OFF GGML_OPENMP=OFF)
target_include_directories(${CMAKE_PROJECT_NAME} PRIVATE ${LLAMA_SRC} ${LLAMA_SRC}/common ${LLAMA_SRC}/include ${LLAMA_SRC}/ggml/include ${LLAMA_SRC}/ggml/src)
target_link_libraries(${CMAKE_PROJECT_NAME} llama llama-common android log)
